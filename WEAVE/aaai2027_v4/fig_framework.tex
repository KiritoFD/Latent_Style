% fig_framework.tex -- TikZ architecture diagram for WEAVE (Figure 2)
% Method (matches paper.tex): endpoint (final-step) per-subband AdaIN.
% Deliberately EXCLUDES: Heun, WCT-as-separate module, Semantic Regional SWD,
% K-means, Quantile Matching, and any "per-step" wording.
\begin{tikzpicture}[
    >=Stealth,
    box/.style   = {draw, rounded corners=3pt, thick, align=center,
                    minimum height=1.0cm, font=\small},
    sub/.style   = {draw, rounded corners=2pt, thick, align=center,
                    minimum width=2.0cm, minimum height=0.7cm, font=\scriptsize, fill=green!8},
    head/.style  = {draw, rounded corners=2pt, align=center,
                    minimum width=2.0cm, minimum height=0.62cm, font=\scriptsize, fill=blue!8},
    nohead/.style= {draw, rounded corners=2pt, align=center,
                    minimum width=2.0cm, minimum height=0.62cm, font=\scriptsize, fill=red!8},
    net/.style   = {draw, rounded corners=6pt, dashed, thick},
    note/.style  = {align=center, font=\scriptsize, text=black!75},
    arrow/.style = {->, thick},
  ]

  % ============ Stage 1: input + frozen VAE encoder ============
  \node[box, text width=2.0cm] (src) at (0,0)    {content\\image $x_c$};
  \node[box, text width=2.5cm] (enc) at (2.8,0)  {SD1.5 VAE\\Encoder $\mathcal E$\\(frozen, $\sim$84M)};
  \node[box, text width=2.0cm] (z0)  at (5.6,0)  {latent $z_0$\\$4{\times}32{\times}32$};
  \draw[arrow] (src) -- (enc);
  \draw[arrow] (enc) -- (z0);

  % ============ Stage 2: single-level Haar DWT ============
  \node[box, text width=2.5cm, fill=orange!10] (dwt) at (8.4,0)
        {Single-level\\Haar DWT\\(orthogonal)};
  \draw[arrow] (z0) -- (dwt);

  % ============ Stage 3: subbands (vertical stack) ============
  \node[sub] (LL) at (11.3, 2.0) {LL $(\ell)$\\$4{\times}16{\times}16$};
  \node[sub] (LH) at (11.3, 0.7) {LH $(h_1)$\\$4{\times}16{\times}16$};
  \node[sub] (HL) at (11.3,-0.7) {HL $(h_2)$\\$4{\times}16{\times}16$};
  \node[sub] (HH) at (11.3,-2.0) {HH $(h_3)$\\$4{\times}16{\times}16$};
  \draw[arrow] (dwt) -- (LL);
  \draw[arrow] (dwt) -- (LH);
  \draw[arrow] (dwt) -- (HL);
  \draw[arrow] (dwt) -- (HH);

  % ============ Stage 3-4: velocity network + 8-step Euler ============
  \node[net, minimum width=3.6cm, minimum height=6.6cm] (netbox) at (14.0,0) {};
  \node[align=center, font=\small\bfseries] at (14.0, 2.95)
        {velocity network $v_\theta$ ($\sim$903K)};
  \node[head]  (vhL) at (14.0, 2.0) {head $v_\ell$};
  \node[head]  (vh1) at (14.0, 0.7) {head $v_{h_1}$};
  \node[head]  (vh2) at (14.0,-0.7) {head $v_{h_2}$};
  \node[nohead] (vh3) at (14.0,-2.0) {$\times$ no head};
  \draw[arrow] (LL) -- (vhL);
  \draw[arrow] (LH) -- (vh1);
  \draw[arrow] (HL) -- (vh2);
  \draw[arrow] (HH) -- (vh3);
  % Euler loop annotation
  \draw[->, thick, rounded corners=3pt, blue!60]
        (14.0,-1.35) -- ++(0,-0.45) -- ++(-1.1,0) -- ++(0,0.45);
  \node[align=center, font=\scriptsize, text=blue!70] at (14.0,-2.55)
        {8-step Euler: $\ell,h_1,h_2$ updated by $v\!\cdot\!dt$;\\$h_3$ carried unchanged};

  % ============ Stage 5: endpoint AdaIN (final step only) ============
  \node[box, text width=2.7cm, fill=blue!10] (adain) at (17.9,0)
        {Endpoint AdaIN\\(once, final step)\\LH:0.3\; HL:0.3\; HH:0.5\\LL excluded (0)};
  \draw[arrow] (netbox.east) -- (adain.west);
  % target style reference
  \node[note] (zstar) at (17.9, 2.7) {target style ref $z^\star$\\(style-memory / pool)};
  \draw[arrow] (zstar) -- (adain);

  % ============ Stage 6: inverse + output ============
  \node[box, text width=1.7cm] (idwt) at (20.5,0) {IDWT};
  \node[box, text width=1.7cm] (z1)   at (22.3,0) {$\hat z_1$};
  \node[box, text width=2.0cm] (dec)  at (24.2,0) {VAE Dec $\mathcal D$};
  \node[box, text width=2.0cm, fill=red!8] (out) at (26.1,0) {stylized $\hat x_s$};
  \draw[arrow] (adain) -- (idwt);
  \draw[arrow] (idwt) -- (z1);
  \draw[arrow] (z1)   -- (dec);
  \draw[arrow] (dec)  -- (out);

  % ============ auxiliary: style conditioning (weak) + cross-attn ============
  \node[note] at (14.0,-3.7)
        {style conditioning $\to$ shared backbone (weak, $\lambda_{LL}{=}0.3$);\\
         style-memory cross-attn (auxiliary, $\Delta$CLIP-S $< 0.001$)};

\end{tikzpicture}
